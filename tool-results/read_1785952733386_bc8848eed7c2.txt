   265→   265→// Анимированный счётчик
   266→   266→function AnimatedCounter({ target, suffix = '', prefix = '' }: { target: number; suffix?: string; prefix?: string }) {
   267→   267→  const [count, setCount] = useState(0)
   268→   268→  const ref = useRef<HTMLDivElement>(null)
   269→   269→  const hasAnimated = useRef(false)
   270→   270→
   271→   271→  useEffect(() => {
   272→   272→    const observer = new IntersectionObserver(
   273→   273→      ([entry]) => {
   274→   274→        if (entry.isIntersecting && !hasAnimated.current) {
   275→   275→          hasAnimated.current = true
   276→   276→          let start = 0
   277→   277→          const duration = 2000
   278→   278→          const increment = target / (duration / 16)
   279→   279→          
   280→   280→          const timer = setInterval(() => {
   281→   281→            start += increment
   282→   282→            if (start >= target) {
   283→   283→              setCount(target)
   284→   284→              clearInterval(timer)
   285→   285→            } else {
   286→   286→              setCount(Math.floor(start))
   287→   287→            }
   288→   288→          }, 16)
   289→   289→        }
   290→   290→      },
   291→   291→      { threshold: 0.5 }
   292→   292→    )
   293→   293→
   294→   294→    if (ref.current) observer.observe(ref.current)
   295→   295→    return () => observer.disconnect()
   296→   296→  }, [target])
   297→   297→
   298→   298→  return (
   299→   299→    <div ref={ref} className="text-4xl md:text-5xl font-black text-gray-900">
   300→   300→      {prefix}{count}{suffix}
   301→   301→    </div>
   302→   302→  )
   303→   303→}
