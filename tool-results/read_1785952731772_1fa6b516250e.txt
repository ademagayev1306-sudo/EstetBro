   265→// Анимированный счётчик
   266→function AnimatedCounter({ target, suffix = '', prefix = '' }: { target: number; suffix?: string; prefix?: string }) {
   267→  const [count, setCount] = useState(0)
   268→  const ref = useRef<HTMLDivElement>(null)
   269→  const hasAnimated = useRef(false)
   270→
   271→  useEffect(() => {
   272→    const observer = new IntersectionObserver(
   273→      ([entry]) => {
   274→        if (entry.isIntersecting && !hasAnimated.current) {
   275→          hasAnimated.current = true
   276→          let start = 0
   277→          const duration = 2000
   278→          const increment = target / (duration / 16)
   279→          
   280→          const timer = setInterval(() => {
   281→            start += increment
   282→            if (start >= target) {
   283→              setCount(target)
   284→              clearInterval(timer)
   285→            } else {
   286→              setCount(Math.floor(start))
   287→            }
   288→          }, 16)
   289→        }
   290→      },
   291→      { threshold: 0.5 }
   292→    )
   293→
   294→    if (ref.current) observer.observe(ref.current)
   295→    return () => observer.disconnect()
   296→  }, [target])
   297→
   298→  return (
   299→    <div ref={ref} className="text-4xl md:text-5xl font-black text-gray-900">
   300→      {prefix}{count}{suffix}
   301→    </div>
   302→  )
   303→}
